cmake_minimum_required(VERSION 3.20)
project(aqua_jolt C CXX)
include(AquariusLibrary)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JoltPhysics-5.3.0.tar.gz" jolt_archive_hash)
if(NOT jolt_archive_hash STREQUAL "e7f9621e480646c434150e1fbe3a9410f4ec4b04ffe54791e0678326b741b918")
    message(FATAL_ERROR "Jolt Physics 5.3.0 source archive checksum mismatch")
endif()
set(PHYSICS_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JoltPhysics-5.3.0")
set(ENABLE_OBJECT_STREAM OFF)
set(CPP_EXCEPTIONS_ENABLED OFF)
set(CPP_RTTI_ENABLED OFF)
set(INTERPROCEDURAL_OPTIMIZATION OFF)
set(CROSS_PLATFORM_DETERMINISTIC ON)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF)
set(ENABLE_ALL_WARNINGS OFF)
set(GENERATE_DEBUG_SYMBOLS OFF)
set(USE_SSE4_1 OFF)
set(USE_SSE4_2 OFF)
set(USE_AVX OFF)
set(USE_AVX2 OFF)
set(USE_LZCNT OFF)
set(USE_TZCNT OFF)
set(USE_F16C OFF)
set(USE_FMADD OFF)
include("${PHYSICS_REPO_ROOT}/Jolt/Jolt.cmake")
# This binding deliberately uses Jolt's single-thread job scheduler on every platform.
set_target_properties(Jolt PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_STANDARD 20
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON DISABLE_PRECOMPILE_HEADERS ON)
if(AQUA_TARGET STREQUAL "web")
    get_target_property(jolt_sources Jolt SOURCES)
    list(FILTER jolt_sources EXCLUDE REGEX "(JobSystemThreadPool|JobSystemWithBarrier|Semaphore)\\.cpp$")
    set_property(TARGET Jolt PROPERTY SOURCES ${jolt_sources})
    set_property(TARGET Jolt PROPERTY COMPILE_OPTIONS "")
    set_property(TARGET Jolt PROPERTY INTERFACE_COMPILE_OPTIONS "")
    set_property(TARGET Jolt PROPERTY INTERFACE_LINK_OPTIONS "")
endif()
aqua_add_library(jolt SOURCES src/jolt.cpp src/aq_json.c src/json_support.c
    INCLUDE_DIRECTORIES src LINK_LIBRARIES Jolt)
set_target_properties(aqua_jolt PROPERTIES VISIBILITY_INLINES_HIDDEN ON)
if(WIN32)
    target_link_options(aqua_jolt PRIVATE -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
endif()
